cmake_minimum_required(VERSION 3.22)
project(TmlAndroidHost C)
if(NOT ANDROID)
  message(FATAL_ERROR "This build requires the Android NDK toolchain")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_ANDROID_JAR OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL ON CACHE BOOL "" FORCE)
set(BUILD_SDL3 ON CACHE BOOL "" FORCE)
set(BUILD_UTILS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FAUDIO_INSTALL OFF CACHE BOOL "" FORCE)
add_link_options("-Wl,-z,max-page-size=16384")
add_subdirectory("${TML_SDL_SOURCE}" sdl)
set(SDL3_INCLUDE_DIRS "${TML_SDL_SOURCE}/include")
set(SDL3_LIBRARIES SDL3::SDL3)
add_subdirectory("${TML_FNA_SOURCE}/lib/FNA3D" fna3d)
target_compile_options(FNA3D PRIVATE -UFNA3D_DRIVER_SDL)
add_subdirectory("${TML_FNA_SOURCE}/lib/FAudio" faudio)
add_library(tmlhost SHARED sdl_entry.c)
target_link_libraries(tmlhost PRIVATE SDL3::SDL3 log)
